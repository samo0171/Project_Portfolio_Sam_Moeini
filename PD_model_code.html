target_col = 'Default next month '
X = cleaned_data.drop(columns=[target_col, "Cust_id"])
y = cleaned_data[target_col]

# 80/20 split
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=1, stratify=y)

# Test av standardmodeller för tabulär data
models = {"LogRegression": Pipeline([('scaler', StandardScaler()), ('clf', LogisticRegression(max_iter=1000))]),
    "RandomForest": RandomForestClassifier(n_estimators=100, n_jobs=-1, random_state=1),
    "GradientBoosting": GradientBoostingClassifier(random_state=1)} # Industristandard för tabulär data som täcker linjära samband och icke linjära mönster

print("Model Comparison (AUC) ")
for name, model in models.items():
    model.fit(X_train, y_train)
    auc = roc_auc_score(y_test, model.predict_proba(X_test)[:, 1])
    print(f"{name}: {auc:.4f}")


print("")
print("Final Model Training (Gradient Boosting)")

# Gradient Boosting
gb_clf = GradientBoostingClassifier(n_estimators=100, learning_rate=0.1, max_depth=3, random_state=1)
gb_clf.fit(X_train, y_train)

y_pred = gb_clf.predict(X_test)
y_prob = gb_clf.predict_proba(X_test)[:, 1]


# Resultat
print(f"ROC-AUC Score: {roc_auc_score(y_test, y_prob):.4f}")
print("")
print("Classification Report:")
print(classification_report(y_test, y_pred))

cm = confusion_matrix(y_test, y_pred)
disp = ConfusionMatrixDisplay(confusion_matrix=cm, display_labels=["Betalar", "Default"])
disp.plot(cmap='Blues')
plt.title(f"Confusion Matrix (AUC: {roc_auc_score(y_test, y_prob):.2f})")
plt.show()

feature_importance = pd.DataFrame({"Feature": X.columns,"Importance": gb_clf.feature_importances_}).sort_values(by='Importance', ascending=False)

print("")
print("Viktigaste variablerna:")
print(feature_importance)

print("Ekonomisk Simulering")

COST_OF_DEFAULT = X_test['x12'].mean()    # Medelvärde x12
COST_OF_ACTION = 50           # Kostnad att bearbeta kunden antar jag är 50
SUCCESS_RATE = 0.20           # Hur stor andel av defaultade kunder vi kan få betalningar av genom bearbetning antar 20%

tn, fp, fn, tp = confusion_matrix(y_test, y_pred).ravel()

# 1: Gör ingenting (Ingen modell)
loss_no_model = (tp + fn) * COST_OF_DEFAULT

# Använd Modellen
intervention_cost = (tp + fp) * COST_OF_ACTION

# De vi missade (False Negatives) förlorar vi helt
loss_missed_customers = fn * COST_OF_DEFAULT

# De vi hittade (True Positives) men misslyckades med att rädda (80% av dem)
loss_failrate_defaults = tp * (1 - SUCCESS_RATE) * COST_OF_DEFAULT

total_loss_with_model = intervention_cost + loss_missed_customers + loss_failrate_defaults

savings = loss_no_model - total_loss_with_model

print("")
print("Förlust utan modell: ", loss_no_model)
print("")
print("Förlust med modell: ", total_loss_with_model)
print("")
print("---------------------------------------------")
print("POTENTIEL BESPARING: ", savings)
print("")
print("ROI: ", (savings / intervention_cost) )
